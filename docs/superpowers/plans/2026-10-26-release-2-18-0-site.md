# 2.18.0: сайт продукта на GitHub Pages — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** одностраничная витрина продукта (ru + en) со скриншотами интерфейса, публикуемая на GitHub Pages при выпуске.

**Architecture:** статический каталог `site/` (HTML + CSS, без JS и зависимостей). Скриншоты снимает локальный сценарий Playwright `pnpm site:shots` против демо-стека и кладёт в `site/shots/<lang>/`; они коммитятся. `pnpm site:check` (Node, без зависимостей) проверяет структуру сайта в CI; workflow `pages.yml` выкладывает `site/` по тегу выпуска.

**Tech Stack:** Node 22 ESM (`node:test`), Playwright 1.63 (`e2e/`), GitHub Actions (`actions/configure-pages`, `upload-pages-artifact`, `deploy-pages`).

**Spec:** `docs/superpowers/specs/2026-10-02-carbone-reports-design.md` §49 (правила перевода — §48.2).

```js
#!/usr/bin/env node
// Проверка сайта продукта (§49.4).
// Использование: node scripts/site-check.mjs [--root <корень репозитория>]
// Проверяет site/: страницы ru и en с переключателем языка, относительные ссылки внутри site/,
// alt/width/height у <img>, одинаковый набор снимков ru/en (= SHOTS) и показ каждого на своей
// странице, размеры файлов, отсутствие <script> и внешних ресурсов, версию в подвале = CHANGELOG.
// Ошибки — «файл[:строка]: …», код выхода 1; неверные аргументы — сообщение в stderr, код 2.
import { existsSync, readdirSync, readFileSync, realpathSync, statSync } from 'node:fs';
import { dirname, join, posix, relative } from 'node:path';
import { fileURLToPath } from 'node:url';

/** Снимки §49.2 (без расширения), одинаковые для всех языков. */
export const SHOTS = [
  'login',
  'reports',
  'run',
  'history',
  'editor-sql',
  'editor-document',
  'signature',
  'access',
  'users',
  'datasources',
];
export const LANGS = ['ru', 'en'];
export const PAGES = { ru: 'index.html', en: 'en/index.html' };
export const MAX_SHOT = 600 * 1024;
export const MAX_SITE = 10 * 1024 * 1024;

const OTHER = { ru: 'en/index.html', en: '../index.html' };

/** Теги с атрибутами и номером строки. */
function tags(html) {
  const out = [];
  for (const m of html.matchAll(/<([a-zA-Z]+)\b([^>]*)>/g)) {
    const attrs = {};
    for (const a of m[2].matchAll(/([\w-]+)\s*=\s*"([^"]*)"/g)) attrs[a[1].toLowerCase()] = a[2];
    out.push({ name: m[1].toLowerCase(), attrs, line: html.slice(0, m.index).split('\n').length });
  }
  return out;
}

const isAbsolute = (url) => /^[a-z][a-z0-9+.-]*:/i.test(url) || url.startsWith('//');

function walk(dir) {
  const files = [];
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    const p = join(dir, e.name);
    if (e.isDirectory()) files.push(...walk(p));
    else files.push(p);
  }
  return files;
}

function changelogVersion(root) {
  const p = join(root, 'CHANGELOG.md');
  if (!existsSync(p)) return null;
  return /^## \[(\d+\.\d+\.\d+)\]/m.exec(readFileSync(p, 'utf8'))?.[1] ?? null;
}

function checkPage(root, lang, version, errors) {
  const rel = `site/${PAGES[lang]}`;
  const file = join(root, rel);
  if (!existsSync(file)) {
    errors.push(`${rel}: нет файла`);
    return;
  }
  const html = readFileSync(file, 'utf8');
  const err = (line, msg) => errors.push(`${rel}:${line}: ${msg}`);
  const all = tags(html);

  // 1. Переключатель — первый элемент после <body>, ссылка на страницу другого языка.
  const body = all.findIndex((t) => t.name === 'body');
  const first = all[body + 1];
  const nav = /<nav class="lang">([\s\S]*?)<\/nav>/.exec(html);
  if (!first || first.name !== 'nav' || first.attrs.class !== 'lang' || !nav)
    err(first?.line ?? 1, 'первым элементом <body> должен быть переключатель <nav class="lang">');
  else if (!nav[1].includes(`href="${OTHER[lang]}"`))
    err(first.line, `переключатель: нет ссылки на ${OTHER[lang]}`);

  // 6. Внешние ресурсы и скрипты.
  for (const t of all) {
    if (t.name === 'script') err(t.line, '<script> на сайте не допускается');
    const res = t.name === 'link' ? t.attrs.href : ['img', 'source', 'iframe'].includes(t.name) ? t.attrs.src : null;
    if (res && isAbsolute(res)) err(t.line, `внешний ресурс ${res}`);
  }

  // 2. Относительные ссылки — существующие файлы внутри site/.
  const siteDir = join(root, 'site');
  const shown = new Set();
  for (const t of all) {
    const url = t.attrs.src ?? t.attrs.href;
    if (url === undefined || url === '' || url.startsWith('#') || isAbsolute(url)) continue;
    const path = decodeURIComponent(url.split(/[?#]/)[0]);
    const target = join(dirname(file), path);
    const inside = relative(siteDir, target);
    if (inside.startsWith('..')) {
      err(t.line, `ссылка ${url} ведёт за пределы site/ (на Pages её нет — нужен абсолютный URL)`);
      continue;
    }
    if (!existsSync(target)) err(t.line, `ссылка ${url}: нет файла site/${inside.split('\\').join('/')}`);
    const shot = /^shots\/([^/]+)\/([^/]+)\.png$/.exec(inside.split('\\').join('/'));
    if (shot && t.name === 'img') {
      if (shot[1] !== lang) err(t.line, `снимок другого языка: ${url}`);
      else shown.add(shot[2]);
    }
  }

  // 3. alt, width, height у <img>.
  for (const t of all.filter((x) => x.name === 'img'))
    for (const a of ['alt', 'width', 'height'])
      if (!t.attrs[a]?.trim()) err(t.line, `<img src="${t.attrs.src ?? ''}">: нет ${a}`);

  // 4. Каждый снимок языка показан на странице.
  for (const s of SHOTS) if (!shown.has(s)) errors.push(`${rel}: снимок не показан на странице: ${s}`);

  // 7. Версия в подвале.
  const v = /data-version="([^"]*)"/.exec(html)?.[1];
  if (!v) errors.push(`${rel}: нет элемента с data-version`);
  else if (version && v !== version) errors.push(`${rel}: версия ${v}, а в CHANGELOG.md — ${version}`);
}

/** Все ошибки сайта root/site; root — корень репозитория. */
export function checkSite(root) {
  const errors = [];
  const siteDir = join(root, 'site');
  if (!existsSync(siteDir)) return ['site: нет каталога'];
  const version = changelogVersion(root);
  if (!version) errors.push('CHANGELOG.md: не найдена версия «## [X.Y.Z]»');
  for (const lang of LANGS) checkPage(root, lang, version, errors);

  // 4. Набор снимков в каждом каталоге = SHOTS.
  for (const lang of LANGS) {
    const dir = join(siteDir, 'shots', lang);
    const have = existsSync(dir) ? readdirSync(dir).filter((f) => !f.startsWith('.')) : [];
    for (const f of have)
      if (!SHOTS.includes(f.replace(/\.png$/, '')) || !f.endsWith('.png'))
        errors.push(`site/shots/${lang}: лишний снимок ${f}`);
    for (const s of SHOTS)
      if (!have.includes(`${s}.png`)) errors.push(`site/shots/${lang}: нет снимка ${s}.png`);
  }

  // 5. Размеры.
  let total = 0;
  for (const f of walk(siteDir)) {
    const size = statSync(f).size;
    total += size;
    const rel = posix.join('site', relative(siteDir, f).split('\\').join('/'));
    if (f.endsWith('.png') && size > MAX_SHOT) errors.push(`${rel}: больше 600 КБ (${Math.ceil(size / 1024)} КБ)`);
  }
  if (total > MAX_SITE) errors.push(`site: больше 10 МБ (${(total / 1048576).toFixed(1)} МБ)`);
  return errors;
}

function main(argv) {
  let root = join(dirname(fileURLToPath(import.meta.url)), '..');
  for (let i = 0; i < argv.length; i++) {
    if (argv[i] === '--root' && argv[i + 1]) root = argv[++i];
    else {
      console.error(`неизвестный аргумент: ${argv[i]}\nИспользование: node scripts/site-check.mjs [--root <каталог>]`);
      process.exitCode = 2;
      return;
    }
  }
  const errors = checkSite(root);
  for (const e of errors) console.log(e);
  if (errors.length) {
    console.log(`site:check — ошибок: ${errors.length}`);
    process.exitCode = 1;
  } else console.log('site:check — ок');
}

if (process.argv[1] && realpathSync(process.argv[1]) === realpathSync(fileURLToPath(import.meta.url)))
  main(process.argv.slice(2));
```

```ts
import { test, expect, type Page } from '@playwright/test';
import { mkdirSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { DEMO } from '../../demo/template';
import { adminApi, demoTemplateId, loginAdminUi, waitForStack } from '../fixtures';

// Снимки экранов для site/ (§49.2, §49.3). Сначала ru, затем en; в конце язык администратора и
// настройки PDF демо-шаблона возвращаются к исходным, даже если съёмка упала.
const OUT = fileURLToPath(new URL('../../site/shots/', import.meta.url));
const GENERATE = { ru: 'Сформировать', en: 'Generate' } as const;

type Lang = keyof typeof GENERATE;

let templateId = '';
let savedLocale: string | null = null;
let savedPdfOptions: Record<string, unknown> | undefined;

test.describe.configure({ mode: 'serial' });

test.beforeAll(async () => {
  await waitForStack();
  const api = await adminApi();
  templateId = await demoTemplateId(api);
  savedLocale = (await api.get<{ locale: string | null }>('/api/auth/me')).locale;
  savedPdfOptions = (
    await api.get<{ pdfOptions: Record<string, unknown> }>(`/api/templates/${templateId}`)
  ).pdfOptions;
  // Для экрана «подпись»: включена подпись PDF методом ГОСТ (пароль не передаётся и не меняется).
  await api.put(`/api/templates/${templateId}/pdf-options`, {
    ...savedPdfOptions,
    signature: { position: 'bottom-right', method: 'gost' },
  });
});

test.afterAll(async () => {
  const api = await adminApi();
  try {
    if (savedPdfOptions) await api.put(`/api/templates/${templateId}/pdf-options`, savedPdfOptions);
  } finally {
    await api.put('/api/me/locale', { locale: savedLocale });
  }
});

async function shot(page: Page, lang: Lang, name: string): Promise<void> {
  await page.evaluate(() => document.fonts.ready);
  await page.screenshot({
    path: `${OUT}${lang}/${name}.png`,
    animations: 'disabled',
    caret: 'hide',
  });
}

for (const lang of ['ru', 'en'] as const) {
  test.describe(lang, () => {
    // Язык браузера определяет язык страницы входа (до входа выбор пользователя неизвестен).
    test.use({ locale: lang === 'ru' ? 'ru-RU' : 'en-US' });

    test(`снимки ${lang}`, async ({ page }) => {
      mkdirSync(`${OUT}${lang}`, { recursive: true });
      const api = await adminApi();
      await api.put('/api/me/locale', { locale: lang });

      // 1. Вход.
      await page.goto('/login');
      await expect(page.getByRole('button', { name: lang === 'ru' ? 'Войти' : 'Sign in' })).toBeVisible();
      await shot(page, lang, 'login');

      // 2. Каталог отчётов.
      await loginAdminUi(page);
      await expect(page.getByText(DEMO.templateName, { exact: true }).first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'reports');

      // 3. Запуск: форма параметров с значениями по умолчанию (зависимые SQL-списки загружены).
      await page.getByText(DEMO.templateName, { exact: true }).first().click();
      await expect(page.getByRole('combobox').first()).toContainText('ООО «Ромашка»');
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'run');

      // 4. История: сначала запуск отчёта, чтобы в истории была свежая запись.
      const rendered = page.waitForResponse(
        (r) => /\/api\/reports\/[^/]+\/render$/.test(r.url()) && r.ok(),
        { timeout: 180_000 },
      );
      await page.getByRole('button', { name: GENERATE[lang] }).click();
      await rendered;
      await page.goto('/history');
      await expect(page.getByText(DEMO.templateName, { exact: true }).first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'history');

      // 5. Редактор: запросы и параметры (Monaco).
      await page.goto(`/admin/templates/${templateId}?tab=data`);
      await expect(page.locator('.monaco-editor').first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'editor-sql');

      // 6. Редактор: документ в OnlyOffice (ожидание — как в e2e/tests/onlyoffice.spec.ts).
      await page.goto(`/admin/templates/${templateId}?tab=document`);
      const frame = page.frameLocator('iframe[name="frameEditor"]');
      await expect(frame.locator('#id_viewer')).toBeVisible({ timeout: 180_000 });
      await expect(frame.locator('.asc-loadmask')).toHaveCount(0, { timeout: 120_000 });
      await shot(page, lang, 'editor-document');

      // 7. Настройки PDF и подписи (подпись ГОСТ включена в beforeAll).
      await page.goto(`/admin/templates/${templateId}?tab=pdf`);
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'signature');

      // 8–10. Доступ, пользователи, источники данных.
      await page.goto('/admin/access');
      await expect(page.getByText(DEMO.templateName, { exact: true }).first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'access');

      await page.goto('/admin/users');
      await expect(page.getByRole('table').first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'users');

      await page.goto('/admin/datasources');
      await expect(page.getByText(DEMO.datasourceName, { exact: true }).first()).toBeVisible();
      await page.waitForLoadState('networkidle');
      await shot(page, lang, 'datasources');
    });
  });
}
```

```yaml
name: Сайт

# Сайт продукта (§49.5): выкладка site/ на GitHub Pages при выпуске (тег vX.Y.Z) или вручную.
on:
  push:
    tags: ['v[0-9]+.[0-9]+.[0-9]+']
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  deploy:
    name: Публикация сайта
    runs-on: ubuntu-latest
    timeout-minutes: 10
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - uses: pnpm/action-setup@ea17c68df8912ef543352723c149a84f56e3d413 # v6.1.0
      - uses: actions/setup-node@949feb2413d6458794dcd2491c4babbbce0c15c1 # v7.1.0
        with:
          node-version: 22
      - run: pnpm site:check
      - uses: actions/configure-pages@45bfe0192ca1faeb007ade9deae92b16b8254a0d # v6.0.0
      - uses: actions/upload-pages-artifact@fc324d3547104276b827a68afc52ff2a11cc49c9 # v5.0.0
        with:
          path: site
      - id: deployment
        uses: actions/deploy-pages@368f82528645a54fb793d4d04e342629a3f51346 # v5.0.1
```
